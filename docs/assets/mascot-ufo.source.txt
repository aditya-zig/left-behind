  g.add(mesh(new T.SphereGeometry(1, 32, 24), M(0x0d0f12, { roughness: .1, clearcoat: 1, clearcoatRoughness: .05 }), null, [.17, .21, .1]));
  g.add(mesh(new T.SphereGeometry(.04, 12, 10), new T.MeshBasicMaterial({ color: 0xffffff }), [.05, .09, .085]));

  head.add(g); return g;

/* ===== voice + lip sync ===== */
let ac, an, buf, speaking = false, open = 0, hop = 0;
const bub = document.getElementById('bubble');
async function speak(text) {
  bub.textContent = text; hop = 1; speaking = true; speechSynthesis.cancel();
  try {
    const r = await fetch(VOICE_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ text }) });
    if (!r.ok) throw 0;
    const a = new Audio(URL.createObjectURL(await r.blob()));
    ac = ac || new AudioContext(); const src = ac.createMediaElementSource(a);
    an = ac.createAnalyser(); an.fftSize = 256; src.connect(an); an.connect(ac.destination);
    buf = new Uint8Array(an.frequencyBinCount);
    a.onended = () => { speaking = false; an = null; };
    await a.play();
  } catch (e) {
    an = null; const u = new SpeechSynthesisUtterance(text); u.pitch = 1.5; u.rate = 1.05;
    u.onend = u.onerror = () => speaking = false; speechSynthesis.speak(u);
  }
}
async function reply(q) {
  if (BRAIN_URL) { try { const r = await fetch(BRAIN_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: q }) }); return (await r.json()).reply; } catch (e) {} }
  const k = /\b(hi|hello|hey)\b/i.test(q) ? 'hi' : /who|name/i.test(q) ? 'who' : /help|how/i.test(q) ? 'help' : /bye/i.test(q) ? 'bye' : 'dflt';
  return LINES[k][Math.floor(R() * LINES[k].length)];
}
const inp = document.getElementById('in');
async function send() { const q = inp.value.trim() || 'hi'; inp.value = ''; speak(await reply(q)); }
document.getElementById('go').onclick = send;

ren.domElement.onclick = () => speak(LINES.dflt[Math.floor(R() * LINES.dflt.length)]);

document.getElementById('glb').onclick = () => {
  const root = ship, p = root.position.clone(), r = root.rotation.clone(), hr = head.rotation.clone();
  root.position.set(0, 0, 0); root.rotation.set(0, 0, 0); head.rotation.set(0, 0, 0);
  new T.GLTFExporter().parse(root, res => {
    root.position.copy(p); root.rotation.copy(r); head.rotation.copy(hr);
    const a = document.createElement('a');
    a.href = URL.createObjectURL(new Blob([res], { type: 'model/gltf-binary' }));
    a.download = 'Blip.glb'; a.click();
  }, { binary: true, onlyVisible: true });
};

function resize() {
  const a = innerWidth / innerHeight; cam.aspect = a; ren.setSize(innerWidth, innerHeight);
  const d = Math.max(10, 6.4 / (2 * Math.tan(16 * Math.PI / 180) * a)); cam.position.set(0, 2, d); cam.lookAt(0, .3, 0); cam.updateProjectionMatrix();
}
addEventListener('resize', resize); resize();
const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
let nextBlink = 2, blink = 0; const clock = new T.Clock();
(function loop() {
  requestAnimationFrame(loop);
  const t = clock.getElapsedTime(), k = reduce ? .3 : 1;
  let target = 0;
  if (an) { an.getByteTimeDomainData(buf); let s = 0; for (const v of buf) s += Math.abs(v - 128); target = Math.min(1, s / buf.length / 26); }
  else if (speaking) target = Math.max(0, .5 + .5 * Math.sin(t * 14) * Math.sin(t * 5.3));
  open += (target - open) * .35;
  mouth.visible = open > .08; smile.visible = !mouth.visible;
  mouth.scale.set(.1 - .03 * open, .02 + .13 * open, .04);
  hop *= .94;
  ship.position.y = (Math.sin(t * 1.5) * .12 + Math.abs(Math.sin(hop * 6)) * hop * .3) * k;
  ship.rotation.z += (-look.x * .1 + Math.sin(t * 1.1) * .02 * k - ship.rotation.z) * .06;
  ship.rotation.x += (look.y * .06 - ship.rotation.x) * .06;
  head.rotation.y += (look.x * .5 - head.rotation.y) * .08;
  head.rotation.x += (look.y * .25 - head.rotation.x) * .08;
  head.position.y = 1.0 + open * .02;
  hands.forEach((h, i) => h.position.y = .5 + (speaking ? Math.abs(Math.sin(t * 9 + i * 2)) * .08 : 0));
  bulbs.forEach((m, i) => m.emissiveIntensity = .5 + .9 * Math.max(0, Math.sin(t * (speaking ? 9 : 2.2) - i * .7)));
  beam.material.opacity = .1 + (speaking ? .08 + open * .1 : 0) + Math.sin(t * 2) * .015;
  stars.rotation.z = t * .004 * k;
  if (t > nextBlink) { blink = 1; nextBlink = t + 2.5 + R() * 3; }
  blink = Math.max(0, blink - .12);
  eyes.forEach(e => e.scale.y = 1 - .85 * Math.sin(blink * Math.PI));
  ren.render(scene, cam);
})();
